package fu.se686868.reservation.config;

import fu.se686868.reservation.dto.*;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.*;

/** Feign synchronous: Reservation -> Room qua HTTP, không gọi thẳng RoomRepository. */
// room.service.url trỏ đến URL của bên bị gọi - khai báo trong file application.properties của bên đi gọi

@FeignClient(name = "roomClient", url = "${room.service.url}")
public interface RoomClient {
    @GetMapping("/api/rooms/{id}")
    ApiResponseDTO findRoom(@PathVariable("id") long id);
}


@SpringBootApplication
@EnableFeignClients                         //QUAN TRỌNG!!! THIẾU LÀ BỊ BÁO LỖI BÊN Client class
public class Se686868ReservationServiceApplication {

    public static void main(String[] args) {
        SpringApplication.run(Se686868ReservationServiceApplication.class, args);
    }
}
